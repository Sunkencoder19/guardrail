function usePrefersReducedMotion() {
  const [reduced, setReduced] = useState(false);
  useEffect(() => {
    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
    setReduced(mq.matches);
    const handler = (e) => setReduced(e.matches);
    mq.addEventListener("change", handler);
    return () => mq.removeEventListener("change", handler);
  }, []);
  return reduced;
}

// Counts up from 0 to `value` once, on mount / whenever value changes.
function useAnimatedNumber(value, duration = 900) {
  const reducedMotion = usePrefersReducedMotion();
  const [display, setDisplay] = useState(reducedMotion ? value : 0);
  const startRef = useRef(null);

  useEffect(() => {
    if (reducedMotion) {
      setDisplay(value);
      return;
    }
    startRef.current = null;
    let raf;
    const step = (timestamp) => {
      if (startRef.current === null) startRef.current = timestamp;
      const elapsed = timestamp - startRef.current;
      const progress = Math.min(1, elapsed / duration);
      const eased = 1 - Math.pow(1 - progress, 3);
      setDisplay(Math.round(eased * value));
      if (progress < 1) raf = requestAnimationFrame(step);
    };
    raf = requestAnimationFrame(step);
    return () => cancelAnimationFrame(raf);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [value, duration, reducedMotion]);

  return display;
}

function buildHeartbeatPath(repeats, unit = 100) {
  let d = "M0,24";
  for (let i = 0; i < repeats; i++) {
    const o = i * unit;
    d += ` L${o + 30},24 L${o + 36},8 L${o + 42},40 L${o + 48},24 L${o + unit},24`;
  }
  return d;
}

function Heartbeat({ reducedMotion, color }) {

    <div className="pointer-events-none absolute inset-x-0 bottom-0 h-12 overflow-hidden opacity-70">
      <div className={reducedMotion ? "" : "animate-heartbeat"} style={{ width: "1200px" }}>
        <svg width="1200" height="48" viewBox="0 0 1200 48" fill="none">

            d={unitPath}
            stroke={color}
            strokeWidth="1.5"

/* ============================================================
   TrendChart — hand-rolled SVG line chart, no dependency,
   draws itself in on mount
============================================================ */

function TrendChart({ data, labels }) {
  const reducedMotion = usePrefersReducedMotion();
  const pathRef = useRef(null);
  const [length, setLength] = useState(0);
  const [drawn, setDrawn] = useState(false);

  const width = 600;
  const height = 180;
  const padding = 12;
  const max = Math.max(...data);
  const min = Math.min(...data);
  const range = max - min || 1;

  const points = data.map((v, i) => {
    const x = padding + (i / (data.length - 1)) * (width - padding * 2);
    const y = padding + (1 - (v - min) / range) * (height - padding * 2);
    return [x, y];
  });

  const linePath = points.map((p, i) => `${i === 0 ? "M" : "L"}${p[0]},${p[1]}`).join(" ");
  const areaPath = `${linePath} L${points[points.length - 1][0]},${height - padding} L${points[0][0]},${height - padding} Z`;

  useEffect(() => {
    if (pathRef.current) setLength(pathRef.current.getTotalLength());
  }, [data]);

  useEffect(() => {
    if (reducedMotion || !length) {
      setDrawn(true);
      return;
    }
    const raf = requestAnimationFrame(() => setDrawn(true));
    return () => cancelAnimationFrame(raf);
  }, [length, reducedMotion]);

  const improving = data[data.length - 1] < data[0];

  return (
    <div className="rounded-xl border border-[#20242E] bg-[#12151C] p-6 h-full">
      <div className="flex items-center justify-between mb-4">
        <div>
          <h3 className="text-[15px] text-[#F5F6F8]" style={{ fontFamily: FONT.body, fontWeight: 500 }}>
            Security Trend
          </h3>
          <p className="text-[12.5px] text-[#8B92A0]" style={{ fontFamily: FONT.body }}>
            Open findings, last 30 days
          </p>
        </div>
        <span
          className="text-[12px] rounded-full px-2.5 py-1"
          style={{
            fontFamily: FONT.mono,
            color: improving ? STATUS.healthy : STATUS.critical,
            backgroundColor: improving ? `${STATUS.healthy}1A` : `${STATUS.critical}1A`,
          }}
        >
          {improving ? "↓ Improving" : "↑ Worsening"}
        </span>
      </div>

      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-40">
        <defs>
          <linearGradient id="trendFill" x1="0" y1="0" x2="0" y2="1">
            <stop offset="0%" stopColor="#4C6FFF" stopOpacity="0.35" />
            <stop offset="100%" stopColor="#4C6FFF" stopOpacity="0" />
          </linearGradient>
        </defs>
        <path d={areaPath} fill="url(#trendFill)" opacity={drawn ? 1 : 0} style={{ transition: "opacity 0.6s ease 0.4s" }} />
        <path
          ref={pathRef}
          d={linePath}
          fill="none"
          stroke="#4C6FFF"
          strokeWidth="2"
          strokeLinecap="round"
          strokeLinejoin="round"
          style={{
            strokeDasharray: length,
            strokeDashoffset: drawn ? 0 : length,
            transition: reducedMotion ? "none" : "stroke-dashoffset 1.4s ease-out",
          }}
        />
      </svg>

      {labels && (
        <div className="mt-2 flex justify-between text-[11px] text-[#565D6D]" style={{ fontFamily: FONT.mono }}>
          <span>{labels[0]}</span>
          <span>{labels[Math.floor(labels.length / 2)]}</span>
          <span>{labels[labels.length - 1]}</span>
        </div>
      )}
    </div>
  );
}
